원리 1 · 피로 신호 차단

아데노신 수용체의 ‘위장 취업’

뇌가 활동하면 피로 물질인 아데노신이 수용체에 결합해 부교감신경을 활성화(휴식 유도)합니다. 하지만 카페인은 아데노신과 입체 구조가 비슷해 수용체에 먼저 결합하여 피로 브레이크를 차단합니다.

원리 2 · 교감신경 흥분

동방결절과 노르에피네프린

브레이크가 풀리면 연수(심장 박동 조절 중추)를 통해 교감신경이 흥분합니다. 교감신경 신경절 이후 뉴런 말단에서 분비된 노르에피네프린이 동방결절(SA node)을 자극해 박동 주기를 짧게 만듭니다.

원리 3 · 효소와 길항 작용

왜 사람마다 그래프가 다를까?

간에서 카페인을 분해하는 효소(CYP1A2)가 풍부한 둔감형은 카페인을 빨리 해독하고 부교감신경(아세틸콜린 분비)이 심박수를 원래대로 낮춥니다. 반면 민감형은 고박동 상태가 수 시간 지속됩니다.

/**
 * [고2 생명과학Ⅰ] '심박동 보고서 제출' 스프레드시트 자동 구성 및 웹 앱 연동 스크립트
 * 열 구조: A열(제출시간) | B열(학년) | C열(반) | D열(번호) | E열(이름) | F열(객관식 점수) | G열(서술형 문항)
 */
const SPREADSHEET_TITLE = '심박동 보고서 제출';
const SHEET_NAME = '심박동 보고서 제출';

// 1. 최초 1회 실행 시 '심박동 보고서 제출' 시트 및 A~G열 헤더를 자동 생성하는 함수
function setupHeartbeatSpreadsheet() {
  let ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    ss = SpreadsheetApp.create(SPREADSHEET_TITLE);
  } else {
    ss.rename(SPREADSHEET_TITLE);
  }
  const sheet = getOrCreateReportSheet_(ss);
  Logger.log('스프레드시트 준비 완료: ' + ss.getUrl());
}

// 2. 학생이 웹 앱 2번 페이지에서 [제출] 버튼을 눌렀을 때 실행되는 POST 엔드포인트
function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = getOrCreateReportSheet_(ss);

    const payload = JSON.parse(e.postData.contents);
    const submitTime = payload.submitTime || Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');

    // A열 ~ G열 순서대로 행 추가
    sheet.appendRow([
      submitTime,          // A열 : 제출시간
      payload.grade,       // B열 : 학년
      payload.classNum,    // C열 : 반
      payload.studentNum,  // D열 : 번호
      payload.name,        // E열 : 이름
      payload.mcScore,     // F열 : 객관식 점수 (5점 만점)
      payload.essayText    // G열 : 서술형 문항
    ]);

    return ContentService.createTextOutput(JSON.stringify({ result: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// 시트 및 A열~G열 헤더 자동 생성 보조 함수
function getOrCreateReportSheet_(ss) {
  let sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  sheet.setName(SHEET_NAME);

  if (sheet.getLastRow() === 0) {
    const headers = ['제출시간', '학년', '반', '번호', '이름', '객관식 점수', '서술형 문항'];
    sheet.appendRow(headers);
    const headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setBackground('#1E4D3B').setFontColor('#FAF7F0').setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}
